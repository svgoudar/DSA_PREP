Problem 167: Two Sum II – Input Array Is Sorted#

Input

  • numbers: List of integers sorted in non-decreasing order

  • target: Integer target sum

Output

  • Indices of the two numbers such that they add up to target

Example

Input: numbers = [2,7,11,15], target = 9
Output: [1,2]
Solution Code 
class Solution:
    def twoSum(self, numbers: List[int], target: int) -> List[int]:
        left, right = 0, len(numbers) - 1
        while left < right:
            total = numbers[left] + numbers[right]
            if total == target:
                return [left + 1, right + 1]
            elif total < target:
                left += 1
            else:
                right -= 1
Complexity Analysis

Time: O(n)
Space: O(1)

Problem 350: Intersection of Two Arrays II#

Input

  • nums1: List of integers

  • nums2: List of integers

Output

  • Intersection of two arrays including duplicates

Example

Input: nums1 = [1,2,2,1], nums2 = [2,2]
Output: [2,2]
Solution Code 
from collections import Counter

class Solution:
    def intersect(self, nums1: List[int], nums2: List[int]) -> List[int]:
        counts = Counter(nums1)
        result = []
        for num in nums2:
            if counts[num] > 0:
                result.append(num)
                counts[num] -= 1
        return result
Complexity Analysis

Time: O(n + m)
Space: O(min(n, m))

Problem 2256: Quickest Way to Lose a Sum#

Input

  • nums: List of integers

Output

  • The index that minimizes the absolute difference between prefix and suffix sums.

Example

Input: nums = [2,3,1,5,4]
Output: 3
Solution Code 
class Solution:
    def minimumAverageDifference(self, nums: List[int]) -> int:
        total = sum(nums)
        min_diff = float('inf')
        res = 0
        prefix = 0

        for i in range(len(nums)):
            prefix += nums[i]
            left_avg = prefix // (i + 1)
            right_avg = 0 if i == len(nums) - 1 else (total - prefix) // (len(nums) - i - 1)
            diff = abs(left_avg - right_avg)
            if diff < min_diff:
                min_diff = diff
                res = i
        return res
Complexity Analysis

Time: O(n)
Space: O(1)

Problem 713: Subarray Product Less Than K#

Input

  • nums: List of positive integers

  • k: Target product

Output

  • Number of contiguous subarrays where the product is less than k

Example

Input: nums = [10, 5, 2, 6], k = 100
Output: 8
Solution Code 
class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 0
        prod = 1
        left = 0
        count = 0
        for right, val in enumerate(nums):
            prod *= val
            while prod >= k:
                prod //= nums[left]
                left += 1
            count += right - left + 1
        return count

Problem 941: Valid Mountain Array#

  • arr: List of integers

Output

  • Return True if it’s a valid mountain array, else False

Example

Input: arr = [0,3,2,1]
Output: True
Solution Code 
class Solution:
    def validMountainArray(self, arr: List[int]) -> bool:
        n = len(arr)
        if n < 3:
            return False
        i = 0
        while i + 1 < n and arr[i] < arr[i + 1]:
            i += 1
        if i == 0 or i == n - 1:
            return False
        while i + 1 < n and arr[i] > arr[i + 1]:
            i += 1
        return i == n - 1

Problem 862: Smallest Subarray with Product at Least K#

Input

  • nums: List of positive integers

  • k: Integer, the target minimum product

Output

  • Length of the smallest subarray such that the product is greater than or equal to k. Return 0 if no such subarray exists.

Example

Input: nums = [10, 5, 2, 6], k = 100
Output: 3
Explanation: The subarray [10, 5, 2] has a product of 100.
Solution Code 
class Solution:
    def minSubArrayLenWithProductAtLeastK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 1

        product = 1
        left = 0
        min_len = float('inf')

        for right in range(len(nums)):
            product *= nums[right]
            while product >= k:
                min_len = min(min_len, right - left + 1)
                product //= nums[left]
                left += 1

        return min_len if min_len != float('inf') else 0
Complexity Analysis

Time: O(n)
Space: O(1)

        curr_sum = 0

Problem 992: Subarrays with K Different Integers#

Input

  • nums: List of integers

  • k: Target number of unique integers

Output

  • Number of subarrays with exactly k distinct integers

Example

Input: nums = [1,2,1,2,3], k = 2
Output: 7
Solution Code 
from collections import defaultdict

class Solution:
    def subarraysWithKDistinct(self, nums: List[int], k: int) -> int:
        def atMostK(k):
            count = defaultdict(int)
            res = left = 0
            for right in range(len(nums)):
                if count[nums[right]] == 0:
                    k -= 1
                count[nums[right]] += 1

                while k < 0:
                    count[nums[left]] -= 1
                    if count[nums[left]] == 0:
                        k += 1
                    left += 1
                res += right - left + 1
            return res

        return atMostK(k) - atMostK(k - 1)
Complexity Analysis

Time: O(n)
Space: O(n)

Problem 259: 3Sum Smaller#

Input

  • nums: List of integers

  • target: Integer target sum

Output

  • Number of index triplets such that the sum is less than target

Example

Input: nums = [-2,0,1,3], target = 2
Output: 2
Solution Code 
class Solution:
    def threeSumSmaller(self, nums: List[int], target: int) -> int:
        nums.sort()
        count = 0
        n = len(nums)

        for i in range(n - 2):
            left, right = i + 1, n - 1
            while left < right:
                if nums[i] + nums[left] + nums[right] < target:
                    count += right - left
                    left += 1
                else:
                    right -= 1
        return count
Complexity Analysis

Time: O(n²)
Space: O(1)

Problem 904: Total Fruit#

Input

  • fruits: List of integers representing fruit types on trees

Output

  • Maximum number of fruits you can collect in two baskets (only two distinct types allowed)

Example

Input: fruits = [1,2,1,2,3]
Output: 4
Solution Code 
from collections import defaultdict

class Solution:
    def totalFruit(self, fruits: List[int]) -> int:
        count = defaultdict(int)
        left = max_len = 0

        for right in range(len(fruits)):
            count[fruits[right]] += 1

            while len(count) > 2:
                count[fruits[left]] -= 1
                if count[fruits[left]] == 0:
                    del count[fruits[left]]
                left += 1

            max_len = max(max_len, right - left + 1)

        return max_len
Complexity Analysis

Time: O(n)
Space: O(1) — max 2 types in map

Problem 974: Longest Subarray With Sum Divisible by K#

Input

  • nums: List of integers

  • k: Integer divisor

Output

  • Length of the longest subarray with a sum divisible by k

Example

Input: nums = [4,5,0,-2,-3,1], k = 5
Output: 5
Solution Code 
class Solution:
    def subarraysDivByK(self, nums: List[int], k: int) -> int:
        mod_map = {0: -1}
        prefix = 0
        max_len = 0

        for i, num in enumerate(nums):
            prefix += num
            mod = prefix % k
            if mod < 0:

        return max_len

Problem 2488: Count Subarrays With Median K#

Input

  • nums: List of integers

  • k: Target median value

Output

  • Number of subarrays where the median is exactly k

Example

Input: nums = [3,2,1,4,5], k = 4
Output: 3
Solution Code 
from collections import Counter

class Solution:
    def countSubarrays(self, nums: List[int], k: int) -> int:
        prefix = Counter()
        i = nums.index(k)
        balance = 0
        prefix[0] = 1
        res = 0

        for j in range(i - 1, -1, -1):
            balance += 1 if nums[j] < k else -1
            prefix[balance] += 1

        balance = 0
        for j in range(i, len(nums)):
            balance += 1 if nums[j] > k else -1 if nums[j] < k else 0
            res += prefix[-balance] + prefix[-balance + 1]

        return res
Complexity Analysis

Time: O(n)
Space: O(n)

Problem 862: Shortest Subarray with Sum at Least K#

Input

  • nums: List of integers (can be negative)

  • k: Target sum

Output

  • Length of the shortest non-empty subarray with sum at least k

Example

Input: nums = [2, -1, 2], k = 3
Output: 3
Solution Code 
from collections import deque

class Solution:
    def shortestSubarray(self, nums: List[int], k: int) -> int:
        n = len(nums)
        prefix = [0] * (n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + nums[i]
        
        dq = deque()
        res = n + 1

        for i in range(n + 1):
            while dq and prefix[i] - prefix[dq[0]] >= k:
                res = min(res, i - dq.popleft())
            while dq and prefix[i] <= prefix[dq[-1]]:
                dq.pop()
            dq.append(i)

        return res if res <= n else -1
Complexity Analysis

Time: O(n)
Space: O(n)

Problem 363: Max Sum of Rectangle No Larger Than K#

Input

  • matrix: 2D list of integers

  • k: Integer target

Output

  • Maximum sum of a rectangle in the matrix such that the sum is no larger than k

Example

Input: matrix = [[1,0,1],[0,-2,3]], k = 2
Output: 2
Solution Code 
import bisect

class Solution:
    def maxSumSubmatrix(self, matrix: List[List[int]], k: int) -> int:
        if not matrix:
            return 0
        max_sum = float('-inf')
        rows, cols = len(matrix), len(matrix[0])

        for left in range(cols):
            row_sums = [0] * rows
            for right in range(left, cols):
                for r in range(rows):
                    row_sums[r] += matrix[r][right]
                
                prefix = [0]
                curr_sum = 0
                for sum_val in row_sums:
                    curr_sum += sum_val
                    idx = bisect.bisect_left(prefix, curr_sum - k)
                    if idx < len(prefix):
                        max_sum = max(max_sum, curr_sum - prefix[idx])
                    bisect.insort(prefix, curr_sum)
        return max_sum
Complexity Analysis

Time: O(cols² × rows × log(rows))
Space: O(rows)

Problem 727: Minimum Window Subsequence#

Input

  • s1: String to search in

  • s2: Subsequence to find

Output

  • Minimum window in s1 which contains s2 as a subsequence

Example

Input: s1 = "abcdebdde", s2 = "bde"
Output: "bcde"
Solution Code 
class Solution:
    def minWindow(self, s1: str, s2: str) -> str:
        m, n = len(s1), len(s2)
        min_len = float('inf')
        start = -1

        i = 0
        while i < m:
            if s1[i] == s2[0]:
                j = 0
                k = i
                while k < m:
                    if s1[k] == s2[j]:
                        j += 1
                        if j == n:
                            end = k + 1
                            j -= 1
                            while k >= i:
                                if s1[k] == s2[j]:
                                    j -= 1
                                    if j < 0:
                                        break
                                k -= 1
                            if end - k - 1 < min_len:
                                min_len = end - k - 1
                                start = k + 1
                            break
                    k += 1
            i += 1

        return "" if start == -1 else s1[start:start + min_len]
Complexity Analysis

Time: O(m × n)
Space: O(1)

Problem 325: Subarray Sum Equals K II#

Input

  • nums: List of integers

  • k: Target sum

Output

  • Length of the longest subarray with sum equal to k

Example

Input: nums = [1, -1, 5, -2, 3], k = 3
Output: 4
Explanation: The subarray [1, -1, 5, -2] sums to 3 and has the longest length.
Solution Code 
class Solution:
    def maxSubArrayLen(self, nums: List[int], k: int) -> int:
        prefix_sum = 0
        max_len = 0
        seen = {0: -1}
        for i, num in enumerate(nums):
            prefix_sum += num
            if prefix_sum - k in seen:
                max_len = max(max_len, i - seen[prefix_sum - k])
            if prefix_sum not in seen:

        return max_len
Complexity Analysis

Problem 548: Split Array With Equal Sum#

Input

  • nums: List of integers

Output

  • Boolean indicating if it’s possible to split array into 4 parts with equal sums

Example

Input: nums = [1,2,1,2,1,2,1]
Output: True
Solution Code 
class Solution:
    def splitArray(self, nums: List[int]) -> bool:
        n = len(nums)
        for j in range(3, n - 3):
            seen = set()
            left_sum = sum(nums[:j])
            temp_sum = 0
            for i in range(1, j - 1):
                temp_sum += nums[i - 1]
                if temp_sum == left_sum - temp_sum - nums[i]:
                    seen.add(temp_sum)
            temp_sum = 0
            for k in range(j + 2, n - 1):
                temp_sum += nums[k - 1]
                if temp_sum == left_sum - temp_sum - nums[k] and temp_sum in seen:
                    return True
        return False
Complexity Analysis

Time: O(n²)
Space: O(n)

Problem 2861: Beautiful Subarrays#

Problem 327: Count of Range Sum#

Input

  • nums: List of integers

  • lower, upper: Integer range

Output

  • Count of subarrays whose sum lies in [lower, upper]

Example

Input: nums = [-2,5,-1], lower = -2, upper = 2
Output: 3
Solution Code 
class Solution:
    def countRangeSum(self, nums: List[int], lower: int, upper: int) -> int:
        from bisect import bisect_left, bisect_right, insort
        prefix = [0]
        total = 0
        count = 0
        for num in nums:
            total += num
            left = total - upper
            right = total - lower
            count += bisect_right(prefix, right) - bisect_left(prefix, left)
            insort(prefix, total)
        return count
Complexity Analysis

Time: O(n log n)
Space: O(n)

Problem 238: Range Product Except Self#

Input

Output

  • Return an array where res[i] is the product of all elements of nums except nums[i]

Example

Input: nums = [1,2,3,4]
Output: [24,12,8,6]
Solution Code 
class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        res = [1] * n

        # Left product
        left = 1
        for i in range(n):
            res[i] = left
            left *= nums[i]

        # Right product
        right = 1
        for i in range(n - 1, -1, -1):
            res[i] *= right
            right *= nums[i]

        return res
Complexity Analysis

Time: O(n)
Space: O(1) (output array not counted)
