Problem 643: Maximum Average Subarray I#

Input

  • nums: List[int], the array of integers

  • k: int, size of the subarray

Output

  • Maximum average value of a contiguous subarray of length k

Example

Input: nums = [1,12,-5,-6,50,3], k = 4
Output: 12.75
Solution Code 
class Solution:
    def findMaxAverage(self, nums: List[int], k: int) -> float:
        window_sum = sum(nums[:k])
        max_sum = window_sum
        for i in range(k, len(nums)):
            window_sum += nums[i] - nums[i - k]
            max_sum = max(max_sum, window_sum)
        return max_sum / k
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(1)

Problem 904: Fruit Into Baskets#

        for right, fruit in enumerate(fruits):
            count[fruit] = count.get(fruit, 0) + 1
            while len(count) > 2:

Problem 239: Sliding Window Maximum#

Input

  • nums: List[int]

  • k: int, window size

Output

  • List of maximums for each sliding window

Example

Input: nums = [1,3,-1,-3,5,3,6,7], k = 3
Output: [3,3,5,5,6,7]
Solution Code 
from collections import deque

class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        dq = deque()
        res = []

        for i in range(len(nums)):
            while dq and dq[0] <= i - k:
                dq.popleft()
            while dq and nums[dq[-1]] < nums[i]:
                dq.pop()
            dq.append(i)
            if i >= k - 1:
                res.append(nums[dq[0]])
        return res
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(k)

Problem 1234: Replace the Substring for Balanced String#

Input

  • s: A string with length divisible by 4 containing only the characters ‘Q’, ‘W’, ‘E’, and ‘R’

Output

  • Minimum length of substring that can be replaced to make the string balanced (all characters appear exactly n/4 times)

Example

Input: s = "QWER"
Output: 0
Solution Code 
from collections import Counter

class Solution:
    def balancedString(self, s: str) -> int:
        n = len(s)
        count = Counter(s)
        res = n
        left = 0

        for right in range(n):
            count[s[right]] -= 1
            while left < n and all(count[c] <= n // 4 for c in "QWER"):
                res = min(res, right - left + 1)
                count[s[left]] += 1
                left += 1
        return res
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(1) – because the count dictionary only holds 4 fixed characters

Problem 20: Valid Parentheses Substring length tracking#

Input

  • s: A string containing just the characters ‘(’, ‘)’, ‘{’, ‘}’, ‘[’ and ‘]’

Output

  • Boolean indicating whether the input string is valid (every open has a matching close in order)

Example

Input: s = "()[]{}"
Output: True
Solution Code 
class Solution:
    def isValid(self, s: str) -> bool:
        stack = []
        mapping = {')': '(', '}': '{', ']': '['}
        for char in s:
            if char in mapping:
                top_element = stack.pop() if stack else '#'
                if mapping[char] != top_element:
                    return False
            else:
                stack.append(char)
        return not stack

Time Complexity: O(n) Space Complexity: O(n)

Problem 438: Find All Anagrams in a String#

Input

  • s: The source string

  • p: The target anagram string

Output

  • List of starting indices of anagrams of p in s

Example

Input: s = "cbaebabacd", p = "abc"
Output: [0, 6]
Solution Code 
from collections import Counter

class Solution:
    def findAnagrams(self, s: str, p: str):
        res = []
        p_count = Counter(p)
        s_count = Counter()

        for i in range(len(s)):
            s_count[s[i]] += 1
            if i >= len(p):
                if s_count[s[i - len(p)]] == 1:
                    del s_count[s[i - len(p)]]
                else:
                    s_count[s[i - len(p)]] -= 1
            if s_count == p_count:
                res.append(i - len(p) + 1)
        return res
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(1) — Because character set is limited to lowercase letters

Problem 76: Minimum Window Substring#

Input

  • s: Source string

  • t: Target string

Output

  • The smallest substring of s containing all characters of t

Example

Input: s = "ADOBECODEBANC", t = "ABC"
Output: "BANC"
Solution Code 
from collections import Counter

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        if not t or not s:
            return ""
        
        count_t = Counter(t)
        window = {}
        have, need = 0, len(count_t)
        res, res_len = [-1, -1], float('inf')
        l = 0

        for r, c in enumerate(s):
            window[c] = window.get(c, 0) + 1
            if c in count_t and window[c] == count_t[c]:
                have += 1
            while have == need:
                if (r - l + 1) < res_len:
                    res = [l, r]
                    res_len = r - l + 1
                window[s[l]] -= 1
                if s[l] in count_t and window[s[l]] < count_t[s[l]]:
                    have -= 1
                l += 1
        l, r = res
        return s[l:r+1] if res_len != float('inf') else ""
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(k) n = length of s, k = length of t

Problem 713: Subarray Product Less Than K#

Input

  • nums: List[int]

  • k: Integer

Output

  • Number of contiguous subarrays where the product of all elements is less than k

Example

Input: nums = [10, 5, 2, 6], k = 100
Output: 8
Solution Code 
class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 0
        prod = 1
        result = left = 0
        for right, val in enumerate(nums):
            prod *= val
            while prod >= k:
                prod //= nums[left]
                left += 1
            result += right - left + 1
        return result
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(1)

Problem 567: Permutation in String#

Input

  • s1: Short string

  • s2: Long string

Output

  • Boolean indicating if any permutation of s1 is a substring of s2

Example

Input: s1 = "ab", s2 = "eidbaooo"
Output: True
Solution Code 
from collections import Counter

class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        if len(s1) > len(s2):
            return False
        s1_count = Counter(s1)
        window = Counter(s2[:len(s1)])

        if window == s1_count:
            return True

        for i in range(len(s1), len(s2)):
            window[s2[i]] += 1
            window[s2[i - len(s1)]] -= 1
            if window[s2[i - len(s1)]] == 0:
                del window[s2[i - len(s1)]]
            if window == s1_count:
                return True

        return False
Complexity Analysis

Time Complexity: O(n) Space Complexity: O(1)

Problem 1763: Longest Nice Substring#

  • s: A string

Example

Input: s = "YazaAay"
Output: "azaA"
Solution Code 
class Solution:
    def longestNiceSubstring(self, s: str) -> str:
        if len(s) < 2:
            return ""
        char_set = set(s)
        for i, c in enumerate(s):
            if c.swapcase() not in char_set:
                left = self.longestNiceSubstring(s[:i])
                right = self.longestNiceSubstring(s[i+1:])
                return left if len(left) >= len(right) else right
        return s

Problem 727: Minimum Window Subsequence#

Input

  • s1: String to search in

  • s2: Subsequence to find

Output

  • Minimum window in s1 which contains s2 as a subsequence

Example

Input: s1 = "abcdebdde", s2 = "bde"
Output: "bcde"
Solution Code 
class Solution:
    def minWindow(self, s1: str, s2: str) -> str:
        m, n = len(s1), len(s2)
        min_len = float('inf')
        start = -1

        i = 0
        while i < m:
            if s1[i] == s2[0]:
                j = 0
                k = i
                while k < m:
                    if s1[k] == s2[j]:
                        j += 1
                        if j == n:
                            end = k + 1
                            j -= 1
                            while k >= i:
                                if s1[k] == s2[j]:
                                    j -= 1
                                    if j < 0:
                                        break
                                k -= 1
                            if end - k - 1 < min_len:
                                min_len = end - k - 1
                                start = k + 1
                            break
                    k += 1
            i += 1

        return "" if start == -1 else s1[start:start + min_len]
Complexity Analysis **Time:** O(m × n), **Space:** O(1)

Problem 30: Substring with Concatenation of All Words#

Input

  • s: The main string

  • words: List of words

Output

  • List of starting indices of substring(s) that are concatenation of each word exactly once

Example

Input: s = "barfoothefoobarman", words = ["foo","bar"]
Output: [0,9]
Solution Code 
from collections import Counter

class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        if not s or not words:
            return []

        word_len = len(words[0])
        num_words = len(words)
        window_len = word_len * num_words
        word_count = Counter(words)
        res = []

        for i in range(word_len):
            left = i
            curr_count = Counter()
            for j in range(i, len(s) - word_len + 1, word_len):
                word = s[j:j + word_len]
                if word in word_count:
                    curr_count[word] += 1
                    while curr_count[word] > word_count[word]:
                        curr_count[s[left:left + word_len]] -= 1
                        left += word_len
                    if j + word_len - left == window_len:
                        res.append(left)
                else:
                    curr_count.clear()
                    left = j + word_len
        return res
Complexity Analysis **Time:** O(n × word_len), **Space:** O(n)

Problem 862: Shortest Subarray with Sum at Least K#

Input

  • nums: List of integers

  • k: Target sum

Output

  • Length of the shortest subarray with sum ≥ k. Return -1 if not found.

Example

Input: nums = [2,-1,2], k = 3
Output: 3
Solution Code 
from collections import deque

class Solution:
    def shortestSubarray(self, nums: List[int], k: int) -> int:
        n = len(nums)
        prefix = [0] * (n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + nums[i]

        dq = deque()
        res = float('inf')

        for i in range(n + 1):
            while dq and prefix[i] - prefix[dq[0]] >= k:
                res = min(res, i - dq.popleft())
            while dq and prefix[i] <= prefix[dq[-1]]:
                dq.pop()
            dq.append(i)

        return res if res != float('inf') else -1
Complexity Analysis

Time: O(n) Space: O(n)

Problem 992: Subarrays with K Different Integers#

Input

  • nums: List of integers

  • k: Target number of unique integers

Output

  • Number of subarrays with exactly k distinct integers

Example

Input: nums = [1,2,1,2,3], k = 2
Output: 7
Solution Code 
from collections import defaultdict

class Solution:
    def subarraysWithKDistinct(self, nums: List[int], k: int) -> int:
        def atMostK(k):
            count = defaultdict(int)
            res = left = 0
            for right in range(len(nums)):
                if count[nums[right]] == 0:
                    k -= 1
                count[nums[right]] += 1

                while k < 0:
                    count[nums[left]] -= 1
                    if count[nums[left]] == 0:
                        k += 1
                    left += 1
                res += right - left + 1
            return res

        return atMostK(k) - atMostK(k - 1)
Complexity Analysis

Time: O(n) Space: O(n)

Problem 1493: Longest Subarray of 1’s After Deleting One Element#

  • Length of the longest subarray with only 1s after deleting exactly one element

Example

Input: nums = [1,1,0,1]
Output: 3
Solution Code 
class Solution:
    def longestSubarray(self, nums: List[int]) -> int:
        left = 0
        zero_count = 0
        max_len = 0

        for right in range(len(nums)):
            if nums[right] == 0:
                zero_count += 1
            while zero_count > 1:
                if nums[left] == 0:
                    zero_count -= 1
                left += 1
            max_len = max(max_len, right - left)

        return max_len
