### ✅ Example

Input: numbers = [2,7,11,15], target = 9

<details>
  <summary><strong>Solution Code</strong></summary>

```python
class Solution:
    def twoSum(self, numbers: List[int], target: int) -> List[int]:
        left, right = 0, len(numbers) - 1
        while left < right:
            total = numbers[left] + numbers[right]
            if total == target:
                return [left + 1, right + 1]
            elif total < target:
                left += 1
            else:
                right -= 1
```

from collections import Counter

class Solution:
    def intersect(self, nums1: List[int], nums2: List[int]) -> List[int]:
        counts = Counter(nums1)
        result = []
        for num in nums2:
            if counts[num] > 0:
                result.append(num)
                counts[num] -= 1
        return result

class Solution:
    def minimumAverageDifference(self, nums: List[int]) -> int:
        total = sum(nums)
        min_diff = float('inf')
        res = 0
        prefix = 0

        for i in range(len(nums)):
            prefix += nums[i]
            left_avg = prefix // (i + 1)
            right_avg = 0 if i == len(nums) - 1 else (total - prefix) // (len(nums) - i - 1)
            diff = abs(left_avg - right_avg)
            if diff < min_diff:
                min_diff = diff
                res = i
        return res

class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 0
        prod = 1
        left = 0
        count = 0
        for right, val in enumerate(nums):
            prod *= val
            while prod >= k:
                prod //= nums[left]
                left += 1
            count += right - left + 1
        return count

        n = len(arr)
        if n < 3:
            return False
        i = 0
        while i + 1 < n and arr[i] < arr[i + 1]:

class Solution:
    def threeSumSmaller(self, nums: List[int], target: int) -> int:
        nums.sort()
        count = 0
        n = len(nums)

        for i in range(n - 2):
            left, right = i + 1, n - 1
            while left < right:
                if nums[i] + nums[left] + nums[right] < target:
                    count += right - left
                    left += 1
                else:
                    right -= 1

from collections import Counter

class Solution:
    def countSubarrays(self, nums: List[int], k: int) -> int:
        prefix = Counter()
        i = nums.index(k)
        balance = 0
        prefix[0] = 1
        res = 0

        for j in range(i - 1, -1, -1):
            balance += 1 if nums[j] < k else -1
            prefix[balance] += 1

        balance = 0
        for j in range(i, len(nums)):
            balance += 1 if nums[j] > k else -1 if nums[j] < k else 0
            res += prefix[-balance] + prefix[-balance + 1]

        return res

### 🧾 Input

* `nums`: List of integers (can be negative)
* `k`: Target sum

### 📤 Output

from collections import deque

class Solution:
    def shortestSubarray(self, nums: List[int], k: int) -> int:
        n = len(nums)
        prefix = [0] * (n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + nums[i]
        
        dq = deque()
        res = n + 1

        for i in range(n + 1):
            while dq and prefix[i] - prefix[dq[0]] >= k:
                res = min(res, i - dq.popleft())
            while dq and prefix[i] <= prefix[dq[-1]]:
                dq.pop()
            dq.append(i)

        return res if res <= n else -1
```

Input: matrix = [[1,0,1],[0,-2,3]], k = 2
Output: 2
```

<details>
  <summary><strong>Solution Code</strong></summary>

```python
import bisect

class Solution:
    def maxSumSubmatrix(self, matrix: List[List[int]], k: int) -> int:
        if not matrix:
            return 0
        max_sum = float('-inf')
        rows, cols = len(matrix), len(matrix[0])

        for left in range(cols):
            row_sums = [0] * rows
            for right in range(left, cols):
                for r in range(rows):
                    row_sums[r] += matrix[r][right]
                
                prefix = [0]
                curr_sum = 0
                for sum_val in row_sums:
                    curr_sum += sum_val
                    idx = bisect.bisect_left(prefix, curr_sum - k)
                    if idx < len(prefix):
                        max_sum = max(max_sum, curr_sum - prefix[idx])
                    bisect.insort(prefix, curr_sum)
        return max_sum
```

  <summary><strong>Complexity Analysis</strong></summary>

**Time:** O(cols² × rows × log(rows))
**Space:** O(rows)

```python
class Solution:
    def minWindow(self, s1: str, s2: str) -> str:
        m, n = len(s1), len(s2)
        min_len = float('inf')
        start = -1

        i = 0
        while i < m:
            if s1[i] == s2[0]:
                j = 0
                k = i
                while k < m:
                    if s1[k] == s2[j]:
                        j += 1
                        if j == n:
                            end = k + 1
                            j -= 1
                            while k >= i:
                                if s1[k] == s2[j]:
                                    j -= 1
                                    if j < 0:
                                        break
                                k -= 1
                            if end - k - 1 < min_len:
                                min_len = end - k - 1
                                start = k + 1
                            break
                    k += 1
            i += 1

        return "" if start == -1 else s1[start:start + min_len]
```

  <summary><strong>Complexity Analysis</strong></summary>

**Time:** O(m × n)

```python
class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        if k <= 1:
            return 0
        prod = 1
        left = 0
        count = 0
        for right, val in enumerate(nums):
            prod *= val
            while prod >= k:
                prod //= nums[left]
                left += 1
            count += right - left + 1
        return count

class Solution:
    def splitArray(self, nums: List[int]) -> bool:
        n = len(nums)
        for j in range(3, n - 3):
            seen = set()
            left_sum = sum(nums[:j])
            temp_sum = 0
            for i in range(1, j - 1):
                temp_sum += nums[i - 1]
                if temp_sum == left_sum - temp_sum - nums[i]:
                    seen.add(temp_sum)
            temp_sum = 0
            for k in range(j + 2, n - 1):
                temp_sum += nums[k - 1]
                if temp_sum == left_sum - temp_sum - nums[k] and temp_sum in seen:
                    return True
        return False

from collections import defaultdict

class Solution:
    def beautifulSubarrays(self, nums: List[int]) -> int:
        xor_map = defaultdict(int)
        xor_map[0] = 1
        xor_val = 0
        count = 0

        for num in nums:
            xor_val ^= num
            count += xor_map[xor_val]
            xor_map[xor_val] += 1

        return count

### 🧾 Input

### 📤 Output

### ✅ Example

```

<details>
  <summary><strong>Solution Code</strong></summary>

```python
class Solution:
    def countRangeSum(self, nums: List[int], lower: int, upper: int) -> int:
        from bisect import bisect_left, bisect_right, insort
        prefix = [0]
        total = 0
        count = 0
        for num in nums:
            total += num
            left = total - upper
            right = total - lower
            count += bisect_right(prefix, right) - bisect_left(prefix, left)
            insort(prefix, total)
        return count
```

### 🧾 Input

### 📤 Output

* Return an array where `res[i]` is the product of all elements of `nums` except `nums[i]`

### ✅ Example

Output: [24,12,8,6]

<details>
  <summary><strong>Solution Code</strong></summary>

```python
class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        res = [1] * n

        # Left product
        left = 1
        for i in range(n):
            res[i] = left
            left *= nums[i]

        # Right product
        right = 1
        for i in range(n - 1, -1, -1):
            res[i] *= right
            right *= nums[i]

        return res
```
